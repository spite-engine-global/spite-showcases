cmake_minimum_required(VERSION 3.25)
project(SpiteShowcases LANGUAGES CXX)
set(SPITE_SDK "" CACHE PATH "Path to SPITE SDK")
find_package(Spite CONFIG REQUIRED PATHS "${SPITE_SDK}/cmake" NO_DEFAULT_PATH)
add_library(WarzoneGame MODULE src/Warzone/WarzoneGame.cpp)
spite_configure_module(WarzoneGame)
target_include_directories(WarzoneGame PRIVATE src/Warzone)
target_link_libraries(WarzoneGame PRIVATE Spite::Base Spite::Ecs Spite::Rendering Spite::World Spite::Physics Spite::RenderRuntime Spite::Control Spite::Serialization Spite::Asset Spite::AssetRuntime Spite::Audio Spite::Application Spite::Ui Spite::UiView)
add_executable(WarzoneTests src/Warzone/tests/WarzoneTests.cpp)
target_compile_features(WarzoneTests PRIVATE cxx_std_20)
target_include_directories(WarzoneTests PRIVATE src/Warzone)
add_library(CityShowcase MODULE src/CityShowcase/CityLogistics.cpp src/CityShowcase/CityPaths.cpp src/CityShowcase/CityPlay.cpp src/CityShowcase/CityShowcase.cpp src/CityShowcase/CitySimulation.cpp src/CityShowcase/CityTransitions.cpp)
add_library(FormationsShowcase MODULE src/FormationsShowcase/FormationsShowcase.cpp)
foreach(target CityShowcase FormationsShowcase)
    spite_configure_module(${target})
    target_link_libraries(${target} PRIVATE Spite::Base Spite::Ecs Spite::Control Spite::Application Spite::Asset Spite::AssetRuntime Spite::Animation Spite::Engine Spite::Localization Spite::Profiling Spite::Rendering Spite::RenderRuntime Spite::Serialization Spite::Ui Spite::UiRuntime Spite::UiView Spite::World)
endforeach()
add_library(UiStorm MODULE src/UiStorm/UiStorm.cpp)
spite_configure_module(UiStorm)
target_link_libraries(UiStorm PRIVATE Spite::Base Spite::Ecs Spite::Control Spite::Serialization Spite::Ui Spite::UiView Spite::UiRuntime Spite::World)
add_executable(CityDesignTests src/CityShowcase/tests/CityDesignTests.cpp)
spite_configure_module(CityDesignTests)
target_link_libraries(CityDesignTests PRIVATE Spite::Base Spite::Ecs)
